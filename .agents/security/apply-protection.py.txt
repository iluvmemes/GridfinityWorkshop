from pathlib import Path
import sys,json
sys.path.insert(0,str(Path('.agents/security').resolve()))
from harden import api,REPO,OUT
results={}
results['actions']=api(REPO+'/actions/permissions','PUT',{'enabled':True,'allowed_actions':'selected','sha_pinning_required':True})
results['allowedActions']=api(REPO+'/actions/permissions/selected-actions','PUT',{'github_owned_allowed':True,'verified_allowed':False,'patterns_allowed':['vimtor/action-zip@*','svenstaro/upload-release-action@*']})
results['mergeSettings']=api(REPO,'PATCH',{'allow_squash_merge':True,'allow_merge_commit':False,'allow_rebase_merge':False,'allow_update_branch':True})
results['mergeSettings']={k:results['mergeSettings'][k] for k in ['allow_squash_merge','allow_merge_commit','allow_rebase_merge','allow_update_branch']}
configs=[
{'name':'Protected integration branches','target':'branch','enforcement':'active','bypass_actors':[],
 'conditions':{'ref_name':{'include':['~DEFAULT_BRANCH','refs/heads/feature/lid-generation'],'exclude':[]}},
 'rules':[{'type':'deletion'},{'type':'non_fast_forward'},{'type':'required_linear_history'},{'type':'required_signatures'},
 {'type':'pull_request','parameters':{'required_approving_review_count':0,'dismiss_stale_reviews_on_push':True,'require_code_owner_review':False,'require_last_push_approval':False,'required_review_thread_resolution':True,'allowed_merge_methods':['squash']}},
 {'type':'required_status_checks','parameters':{'required_status_checks':[{'context':'Security checks','integration_id':15368},{'context':'Dependency review','integration_id':15368}],'strict_required_status_checks_policy':True,'do_not_enforce_on_create':False}},
 {'type':'code_scanning','parameters':{'code_scanning_tools':[{'tool':'CodeQL','alerts_threshold':'errors','security_alerts_threshold':'all'}]}}]},
{'name':'No branch history rewrites','target':'branch','enforcement':'active','bypass_actors':[],
 'conditions':{'ref_name':{'include':['~ALL'],'exclude':[]}},'rules':[{'type':'non_fast_forward'}]},
{'name':'Immutable release tags','target':'tag','enforcement':'active','bypass_actors':[],
 'conditions':{'ref_name':{'include':['refs/tags/v*'],'exclude':[]}},'rules':[{'type':'deletion'},{'type':'non_fast_forward'},{'type':'update','parameters':{'update_allows_fetch_and_merge':False}}]}
]
existing=api(REPO+'/rulesets');byname={r['name']:r['id'] for r in existing}
results['rules']=[]
for config in configs:
 endpoint=REPO+'/rulesets'+('/'+str(byname[config['name']]) if config['name'] in byname else '')
 r=api(endpoint,'PUT' if config['name'] in byname else 'POST',config)
 results['rules'].append(r)
(OUT/'protections.json').write_text(json.dumps(results,indent=2),encoding='utf8')
print(json.dumps({'actions':results['actions'],'rules':[(r['name'],r['id']) for r in results['rules']]},indent=2))
