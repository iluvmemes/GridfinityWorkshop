import json, subprocess
from pathlib import Path
ROOT=Path.cwd();OUT=ROOT/'.agents/security';REPO='repos/iluvmemes/GridfinityWorkshop'
def api(path,method='GET',data=None,optional=False):
    cmd=['gh','api',path,'--method',method]
    if data is not None:cmd+=['--input','-']
    p=subprocess.run(cmd,input=json.dumps(data) if data is not None else None,text=True,encoding='utf8',capture_output=True)
    if p.returncode:
        if optional:return {'unsupported':p.stderr.strip(),'response':p.stdout.strip()}
        raise RuntimeError(path+': '+p.stderr+' '+p.stdout)
    return json.loads(p.stdout) if p.stdout.strip() else {'ok':True}
if __name__=='__main__':
    results={}
    results['alerts']=api(REPO+'/vulnerability-alerts','PUT')
    results['securityUpdates']=api(REPO+'/automated-security-fixes','PUT')
    results['privateReports']=api(REPO+'/private-vulnerability-reporting','PUT')
    for setting in ['secret_scanning','secret_scanning_push_protection','secret_scanning_non_provider_patterns','secret_scanning_validity_checks']:
        response=api(REPO,'PATCH',{'security_and_analysis':{setting:{'status':'enabled'}}},optional=True)
        results[setting]=response if 'unsupported' in response else response.get('security_and_analysis',{}).get(setting)
    results['workflowPermissions']=api(REPO+'/actions/permissions/workflow','PUT',{'default_workflow_permissions':'read','can_approve_pull_request_reviews':False})
    results['forkApproval']=api(REPO+'/actions/permissions/fork-pr-contributor-approval','PUT',{'approval_policy':'all_external_contributors'})
    results['codeql']=api(REPO+'/code-scanning/default-setup','PATCH',{'state':'configured','query_suite':'extended'})
    (OUT/'enabled-settings.json').write_text(json.dumps(results,indent=2),encoding='utf8');print(json.dumps(results,indent=2))
