from pathlib import Path
import sys,json,base64,re,shutil
sys.path.insert(0,str(Path('.agents/security').resolve()))
from harden import api,REPO,OUT
pins={'actions/checkout@v4':'actions/checkout@11d5960a326750d5838078e36cf38b85af677262','actions/setup-python@v5':'actions/setup-python@a26af69be951a213d495a4c3e4e4022e16d87065','actions/setup-node@v4':'actions/setup-node@49933ea5288caeca8642d1e84afbd3f7d6820020','actions/upload-artifact@v4':'actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02','vimtor/action-zip@v1.2':'vimtor/action-zip@1379ea20d4c5705669ba81fd626dd01b1c738f26','svenstaro/upload-release-action@v2':'svenstaro/upload-release-action@29e53e917877a24fad85510ded594ab3c9ca12de'}
def secure_release(s):
    for old,new in pins.items():s=s.replace(old,new+' # '+old.split('@')[1])
    s=s.replace('on:\n','on:\n',1)
    if '\npermissions:\n' not in s:s=s.replace('jobs:\n','permissions:\n  contents: read\njobs:\n')
    else:s=s.replace('permissions:\n  contents: write','permissions:\n  contents: read')
    s=re.sub(r'(    runs-on: [^\n]+\n)',r'\1    permissions:\n      contents: write\n      id-token: write\n      attestations: write\n    timeout-minutes: 20\n',s)
    checkout='      - uses: '+pins['actions/checkout@v4']+' # v4\n'
    s=s.replace(checkout,checkout+'        with:\n          persist-credentials: false\n')
    marker='      - name: Upload the artifacts'
    if marker not in s:marker='      - name: Attach archive to tagged release'
    attest='''      - name: Attest release archive provenance
        if: startsWith(github.ref, 'refs/tags/')
        uses: actions/attest-build-provenance@977bb373ede98d70efdf65b84cb5f73e068dcc2a # v3
        with:
          subject-path: dist/*.zip
'''
    s=s.replace(marker,attest+marker)
    return s
# Stage only security files, preserving uncommitted workshop integration locally.
base=OUT/'files';files={p.relative_to(base).as_posix():p.read_text(encoding='utf8') for p in base.rglob('*') if p.is_file()}
releases={}
for branch in ['master','feature/lid-generation']:
    existing=api(REPO+'/contents/.github/workflows/release.yml?ref='+branch)
    releases[branch]=secure_release(base64.b64decode(existing['content']).decode())
for name,content in files.items():
    p=Path(name);p.parent.mkdir(parents=True,exist_ok=True);p.write_text(content,encoding='utf8')
p=Path('.github/workflows/release.yml');p.write_text(secure_release(p.read_text(encoding='utf8')),encoding='utf8')
# Publish small security-only bootstrap commits, before enabling required-PR protection.
commits={}
for branch in ['master','feature/lid-generation']:
    ref=api(REPO+'/git/ref/heads/'+branch)['object']['sha'];basecommit=api(REPO+'/git/commits/'+ref)
    changes={**files,'.github/workflows/release.yml':releases[branch]}
    tree=api(REPO+'/git/trees','POST',{'base_tree':basecommit['tree']['sha'],'tree':[{'path':name,'mode':'100644','type':'blob','content':content} for name,content in changes.items()]})
    commit=api(REPO+'/git/commits','POST',{'message':'Harden repository security scans, dependency review and release provenance','tree':tree['sha'],'parents':[ref]})
    api(REPO+'/git/refs/heads/'+branch,'PATCH',{'sha':commit['sha'],'force':False})
    commits[branch]={'before':ref,'after':commit['sha'],'files':list(changes)}
(OUT/'bootstrap-commits.json').write_text(json.dumps(commits,indent=2),encoding='utf8')
print(json.dumps(commits,indent=2))
